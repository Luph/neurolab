\hypertarget{README-yeniden-uxfcretme}{%
\section{Yeniden üretme}\label{README-yeniden-uxfcretme}}

\begin{gbkod}
pip install -r kod/requirements.txt
kod/hepsini_uret.sh                                     # bütün hesaplar → veri/ (girdi/'deki ham verilerden)
kod/hepsini_uret.sh --cevrimdisi --cikti /tmp/gb_veri   # ağsız sınama: başka klasöre üret, veri/ ile karşılaştır
latex/derle.sh                                          # şekiller, LaTeX gövdeleri ve bütün PDF'ler → pdf/
\end{gbkod}

\begin{itemize}
\tightlist
\item
  \textbf{Süre.} Hesaplar ≈15--25 dakika sürer; çoğu çekim modelinde (19 PPML spesifikasyonu, 1.000 çekilişli genel denge). PDF'ler ≈4 dakika.
\item
  \textbf{Sınama sonucu (18.10.2026).} Çevrimdışı kipte bütün betikler paketteki girdilerle çalıştı. \kod{veri/\allowbreak{}} klasöründeki sonuçlar bayt bayt aynı üretildi (\kod{kod/\allowbreak{}dogrula.\allowbreak{}py}).
\item
  \textbf{Tek tek çalıştırma.} Betikler \kod{kod/\allowbreak{}} içinden argümansız çalışır. Varsayılan girdi \kod{girdi/\allowbreak{}}, çıktı \kod{veri/\allowbreak{}}.
